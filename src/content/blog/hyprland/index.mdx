## 阻止usb外设唤醒电脑

创建系统服务 `/etc/systemd/system/disable-xhc-wakeup.service`

```ini
[Unit]
Description=Disable all XHC wakeup devices
After=multi-user.target

[Service]
Type=oneshot
ExecStart=/bin/bash -c "grep XHC /proc/acpi/wakeup | awk '{print $1}' | while read dev; do echo $dev | tee /proc/acpi/wakeup; done"
RemainAfterExit=true

[Install]
WantedBy=multi-user.target
```

启用并检查

```bash
sudo systemctl enable --now disable-xhc-wakeup.service
cat /proc/acpi/wakeup | grep XHC # 检查外设是否变为disabled
```

## dotfiles 管理

创建文件夹

创建目录

- home: ~ 目录下的文件, 如 .zshrc

- config: ~/.config 目录下按文件夹分类的文件

- scripts: 存储自定义脚本

- shotcut_scripts: 存储快捷启动脚本

创建链接脚本

功能列表

- 将 home 目录下的文件创建符号链接到用户主目录

- 将 config 目录下的文件夹创建符号链接到 ~/.config 目录

```bash
#!/bin/bash

DOTFILES_DIR=$(cd "$(dirname "$0")"; pwd)

echo "--- 开始应用 Dotfiles ---"

cleanup_dead_links() {
    local target_dir="$1"
    local source_dir="$2"
    echo "清理 $target_dir 中的失效链接..."
    find "$target_dir" -maxdepth 1 -type l | while read -r link; do
        target_path=$(readlink "$link")
        if [[ "$target_path" == "$source_dir"* ]] && [ ! -e "$target_path" ]; then
            rm "$link"
            echo "已删除失效链接: $(basename "$link")"
        fi
    done
}

# 1. 处理 home 文件夹内容
if [ -d "$DOTFILES_DIR/home" ]; then
    for item in "$DOTFILES_DIR/home"/* "$DOTFILES_DIR/home"/.[!.]*; do
        [ -e "$item" ] || continue
        base_name=$(basename "$item")
        target="$HOME/$base_name"
        
        # 核心修改：使用 -snf
        # -s: symbolic
        # -n: no-dereference (防止在目录内创建嵌套链接)
        # -f: force
        ln -snf "$item" "$target"
        echo "已链接: $base_name -> $target"
    done
    cleanup_dead_links "$HOME" "$DOTFILES_DIR/home"
fi

# 2. 处理 config 文件夹内容
if [ -d "$DOTFILES_DIR/config" ]; then
    mkdir -p "$HOME/.config"
    for item in "$DOTFILES_DIR/config"/*; do
        [ -e "$item" ] || continue
        base_name=$(basename "$item")
        target="$HOME/.config/$base_name"
        
        # 同理，使用 -snf
        ln -snf "$item" "$target"
        echo "已链接: $base_name -> $target"
    done
    cleanup_dead_links "$HOME/.config" "$DOTFILES_DIR/config"
fi


echo "--- Dotfiles 应用完成 ---"
```
